0011 | global forcefetch 0   // 1 = ignore local copy, re-fetch everything

0013 | *--- helper: retry a file copy up to 4x with backoff, error out on failure
0014 | cap program drop _fetchcopy
0015 | program define _fetchcopy
0016 |     args label src dst
0017 |     local try 1
0018 |     local ok  0
0019 |     while `try' <= 4 & `ok' == 0 {
0020 |         cap copy "`src'" "`dst'", replace
0021 |         if _rc == 0 local ok 1
0022 |         else {
0023 |             local rc = _rc
0024 |             if `try' < 4 sleep `=3000*`try''
0025 |             local try = `try' + 1
0026 |         }
0027 |     }
0028 |     if `ok' == 0 {
0029 |         dis as error "FAILED: `label' (rc=`rc') after 4 tries. Check the server path/permissions."
0030 |         exit `rc'
0031 |     }
0032 | end
0033 | 
0034 | *--- helper: check-cache-then-copy for one file, sets `status_<tag>' in caller
0035 | cap program drop _fetchone
0036 | program define _fetchone
0037 |     args tag label src dst

0039 |     if $forcefetch == 0 {
0040 |         cap confirm file "`dst'"

0042 |     }
0043 |     if `needfetch' == 0 {
0044 |         dis as text "[`tag'] `label' already in FromServers -- skipping."
0045 |         c_local status`tag' "CACHED"
0046 |     }
0047 |     else {
0048 |         dis as result "[`tag'] Copying `label' ..."
0049 |         _fetchcopy "`tag': `label'" "`src'" "`dst'"
